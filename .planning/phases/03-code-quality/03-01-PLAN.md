export async function move(src: string, dest: string): Promise<void> {
  const destForStat = existsSync(dest) ? dest : dirname(dest)
  if (samefs(src, destForStat)) {
    renameSync(src, dest)
  } else {
    run('cp', ['-a', src, dest])
    run('rm', ['-rf', src])
  }
}
